កិច្ចសន្យាជួល | មូលដ្ឋានគ្រឹះច្បាប់

📜 និយមន័យ

កិច្ចសន្យាដែលភាគីម្ខាងឱ្យភាគីម្ខាងទៀតប្រើប្រាស់ និងអាស្រ័យផលពីវត្ថុណាមួយ ដោយយកថ្លៃ។ មាត្រា ៥៩៦

✍️ ការបង្កើត : ដោយការព្រមព្រៀងគ្នា (លាយលក្ខណ៍អក្សរ ឬផ្ទាល់មាត់)

📌 លក្ខខណ្ឌសំខាន់ៗ៖
វត្ថុជួល + ថ្លៃឈ្នួល + អំឡុងពេល

🏛️ កម្មវត្ថុ

  • ចលនវត្ថុ (រថយន្ត, គ្រឿងម៉ាស៊ីន)
  • អចលនវត្ថុ (ដី, ផ្ទះ, អគារ)

⏳ អំឡុងពេល

មានកំណត់ ឬគ្មានកំណត់ (ម.៥៩៩)
អចលនវត្ថុចាប់ពី ១៥ឆ្នាំឡើង → សិទ្ធិជួលអចិន្ត្រៃយ៍

ម.៥៩៩ ក២, ២៤៤
ផ្អែកតាមក្រមរដ្ឋប្បវេណី | ភតិសន្យា

⚖️ លក្ខខណ្ឌតតាំង & ករណីកិច្ចភតិបតី

🛡️ តតាំងតតិយជន

ភតិកៈអាចតតាំងជាមួយភាគីទីបី បើមានការកាន់កាប់ជាក់ស្ដែងបន្តបន្ទាប់ ម.៥៩៨ ក១

ទាមទារប្រគល់វិញ ឬបញ្ឈប់ការរារាំង បើតតិយជនបំពាន ម.៥៩៨ ក២

🏠 ករណីកិច្ចភតិបតី

✅ ធានាឲ្យភតិកៈប្រើប្រាស់ដោយសេរី គ្មានការរំខាន (ម.៦០០)
✅ ជួសជុលចាំបាច់៖ ការជ្រាបទឹកភ្លៀង ឬការខូចខាតធំ (ម.៦០២)
✅ បើគ្មានលទ្ធភាពជួសជុល → បញ្ចុះថ្លៃជួល ឬបញ្ចប់កិច្ចសន្យា

🧰 ការជួសជុល & សោហ៊ុយចំណាយ
- សោហ៊ុយចាំបាច់ → ភតិបតីត្រូវសងវិញ (ម.៦០៤ ក១)
- សោហ៊ុយបង្កើនតម្លៃ → សងវិញតាមការព្រមព្រៀង ឬតាមច្បាប់ (រយៈពេល១ឆ្នាំ)
ភតិបតី = ម្ចាស់អ្នកឲ្យជួល | ភតិកៈ = អ្នកជួល
🔧🏚️
📋 សិទ្ធិ និងករណីកិច្ចភតិកៈ

💰 ការបង់ថ្លៃឈ្នួល

បង់តាមពេលកំណត់ (ម.៦១០)។ បើគ្មានកំណត់៖
- ចលនវត្ថុ / ផ្ទះ → រាល់ចុងខែ
- ដីធ្លី → រាល់ចុងឆ្នាំ
- វត្ថុតាមរដូវប្រមូលផល → បង់ក្រោយផុតរដូវ

មាត្រា ៦១០ ក២

🔄 សិទ្ធិប្រើប្រាស់

ប្រើប្រាស់តាមវិធីធម្មតា (ម.៦០០) និងថែរក្សាក្នុងនាមអ្នកគ្រប់គ្រងល្អ (ម.៦១៨)
❌ មិនអាចជួលបន្ត ឬអនុប្បទានសិទ្ធិជួល បើគ្មានការយល់ព្រមពីភតិបតី (ម.៦០៨)

📉 សិទ្ធិបន្ថយថ្លៃជួល / រំលាយ
- ករណីចំណូលថយ (ម.៦០៦)
- បាត់បង់ ឬវិនាសវត្ថុជួលមួយផ្នែក (ម.៦០៧)
- ការជួលបន្តមានស្វ័យប្រវត្តិបើគ្មានការជំទាស់ (ម.៦០៩)
ភតិកៈត្រូវប្រើប្រាស់ដោយសុចរិត បើមិនដូច្នោះអាចត្រូវរំលាយកិច្ចសន្យា
💸📑
🚪 ការរំលាយភតិសន្យា

📌 មូលហេតុបញ្ចប់

  • ⏰ ផុតអំឡុងពេលកំណត់ (ម.៦១២)
  • 📢 ស្នើសុំរំលាយដោយគ្មានកំណត់ពេល៖
    ចលនវត្ថុ → ១ថ្ងៃ, អគារ → ៣ខែ, ដីធ្លី → ១ឆ្នាំ (ម.៦១៥)
  • 💥 បាត់បង់/ខូចខាតធ្ងន់ធ្ងរ (ម.៦០៧) បើមានកំហុស → ទទួលខុសត្រូវលើសំណង (ម.៣៩៨)
  • ⚖️ មិនអនុវត្តកាតព្វកិច្ចធ្ងន់ធ្ងរ → រំលាយភ្លាម (ម.៤០៧)

🛠️ វិធីសង្គ្រោះ

បើកូនបំណុលមិនអនុវត្តកាតព្វកិច្ច ម្ចាស់បំណុលអាចប្រើវិធីសង្គ្រោះតាម មាត្រា ៣៩០។

ការរំលាយភតិសន្យាគ្មានអានុភាពប្រតិសកម្ម (ម.៦១៧)។

🔁 ការបញ្ចប់ដោយសារផុតពេល : ភតិកៈសងវត្ថុជួលវិញ ហើយភតិបតីត្រូវបង់ថ្លៃចំណាយបង្កើនតម្លៃ។

ចំណាំសំខាន់ : បើភតិកៈបំពានប្រើប្រាស់មិនសុចរិត ឬជួលបន្តគ្មានការអនុញ្ញាត ភតិបតីអាចរំលាយកិច្ចសន្យាបាន (ម.៦០១ ក២, ម.៦០៨)។

អនុលោមតាមក្រមរដ្ឋប្បវេណី ភាគ៤ ស្តីពីភតិសន្យា
⚖️🔚
📌 សង្ខេបសិទ្ធិ & កាតព្វកិច្ច

👑 ភតិបតី

  • ផ្តល់វត្ថុជួលតាមលក្ខខណ្ឌ
  • ជួសជុលចាំបាច់ (ភាពជ្រាបទឹក ខូចខាតធំ)
  • មិនរំខានការប្រើប្រាស់
  • សងសោហ៊ុយបង្កើនតម្លៃ ឬចាំបាច់
  • ទទួលបានថ្លៃឈ្នួល និងសិទ្ធិរំលាយក្នុងករណីបំពាន

🙋 ភតិកៈ

  • បង់ថ្លៃឈ្នួលទាន់ពេល
  • ប្រើប្រាស់ដោយសុចរិត ថែរក្សាវត្ថុជួល
  • មិនជួលបន្ត ឬផ្ទេរសិទ្ធិដោយគ្មានការអនុញ្ញាត
  • ទាមទារបញ្ចុះថ្លៃ ឬរំលាយនៅពេលមានការខូចខាត/ចំណូលថយ
  • អនុវត្តសិទ្ធិកាន់កាប់តាមផ្លូវច្បាប់

📖 ភតិសន្យាអចលនវត្ថុលើសពី១៥ឆ្នាំ → ចាត់ទុកជាសិទ្ធិជួលអចិន្ត្រៃយ៍ (ម.៥៩៩ ក៣, ២៤៤)

រយៈពេលកំណត់សំខាន់ៗ៖ ការទាមទារសោហ៊ុយ (១ឆ្នាំ) ការជូនដំណឹងរំលាយតាមប្រភេទវត្ថុ

ភតិសន្យា — ឧបករណ៍ច្បាប់សំខាន់សម្រាប់សិទ្ធិប្រើប្រាស់ទ្រព្យសម្បត្តិ
🏛️📜
💡 បញ្ញត្តិពិសេស & យន្តការអនុវត្ត

🏘️ ការជួលអគារលំនៅឋាន

អនុលោមតាមច្បាប់ស្តីពីការជួល បើគ្មានកំណត់ពេល ភាគីត្រូវជូនដំណឹង ៣ខែជាមុន (ម.៦១៥)។ ការដាស់តឿនឱ្យបញ្ឈប់ការបំពានត្រូវកំណត់ពេលសមរម្យ។

⚙️ ការប្រើប្រាស់ខុសពីធម្មតា

បើភតិកៈប្រើប្រាស់វត្ថុជួលខុសពីគោលបំណង ឬធ្វើឱ្យខូចខាត ភតិបតីអាចរំលាយកិច្ចសន្យាបានភ្លាមៗ ដោយមិនចាំបាច់ផ្តល់ពេល (ម.៦០១ ក២ និង ម.៤០៧)។

សិទ្ធិក្នុងការកាន់កាប់ជាក់ស្តែង (ម.៥៩៨, ២៣៦)

🔍 ការតតាំងជាមួយតតិយជន : ភតិកៈអាចប្តឹងទាមទារដោយផ្អែកលើការកាន់កាប់ជាក់ស្តែង (ម.៥៩៨ ក២) និងប្រើប្រាស់សិទ្ធិតាមមាត្រា ២៣៦ ដើម្បីទប់ស្កាត់ការរំខាន។

ភតិសន្យាមិនត្រឹមតែជាកិច្ចសន្យាឯកជន ប៉ុន្តែមានឥទ្ធិពលចំពោះបុគ្គលទីបី បើមានការកាន់កាប់ជាក់ស្តែង
